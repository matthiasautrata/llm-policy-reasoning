cc_anc(C,P) :- cc_parent(C,P).
cc_anc(C,A) :- cc_parent(C,P), cc_anc(P,A).
chain(E,M) :- reports(E,M).

spend_item(C,C,B) :- booked(C,B).
spend_item(A,C,B) :- booked(C,B), cc_anc(C,A).
booked_total(C,T) :- cc(C), T = sum B : { spend_item(C,_,B) }.
headroom(C,H) :- booked_total(C,T), cap(C,K), H = K - T.
over_cap(C) :- booked_total(C,T), cap(C,K), T > K.
charges(R,C) :- req(R,E,_,_,_), emp(E,_,C).
charges(R,A) :- req(R,E,_,_,_), emp(E,_,C), cc_anc(C,A).
cap_breach(R) :- charges(R,C), req(R,_,A,_,_), booked_total(C,T), cap(C,K), T + A > K.
has_approver(R) :- req(R,Q,A,_,_), chain(Q,M), emp(M,Role,_), role_limit(Role,L), L >= A.
vendor_ok(R) :- req(R,_,_,_,V), vendor(V,"approved").
eligible(R) :- has_approver(R), !cap_breach(R), vendor_ok(R).
approved(R) :- eligible(R), !denied(R).
